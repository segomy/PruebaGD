Create DataBase PruebaGD;

use PruebaGD;

Create table Categoria(
    CodigoCategoria int identity(1,1) primary key,
    Nombre varchar(60) not null
)

Create table Producto(
    CodigoProducto int identity(1,1) primary key,
    Nombre varchar(100) not null,
    CodigoCategoria int not null,
    foreign key (CodigoCategoria) references Categoria(CodigoCategoria)
)

Create table Ventas(
    CodigoVenta int identity(1,1) primary key,
    Fecha Date not null,
    CodigoProducto int not null,
    foreign key (CodigoProducto) references Producto(CodigoProducto)
)

select 
    CAST(CONCAT(a.Año, '-', f.MesDia) as DATE) as Fecha,
    p.CodigoProducto
from Años a

cross join Productos p

select top(1) a.fecha,c.Nombre
from Ventas a
    join Producto b on a.codigoproducto = b.codigoproducto
    join Categoria c on c.CodigoCategoria = b.CodigoCategoria
order by a.fecha desc

select c.CodigoCategoria,c.nombre
from Ventas a
    join Producto b on a.codigoproducto = b.codigoproducto
    join Categoria c on c.CodigoCategoria = b.CodigoCategoria
where Year(a.fecha) = 2019 
group by c.CodigoCategoria,c.nombre
order by a.fecha desc
